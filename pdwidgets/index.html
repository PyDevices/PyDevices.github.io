Installation & Minimal Button Loop

import mip
mip.install("pdwidgets", index="https://PyDevices.github.io/mip")

# Example Usage:
import board_config
from pdwidgets.widgets.button import Button

screen = board_config.screen
btn = Button(x=20, y=40, width=120, height=45, text="Click Me")
screen.add(btn)
